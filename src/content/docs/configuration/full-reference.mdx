> **适用版本：技术预览 (v0.5.4)。** 生产环境安全建议：配置文件权限请设为 `0600` (`chmod 600 config.toml`)，避免敏感凭据被本机非特权用户读取。

`frp-tun` 采用标准 TOML 格式进行配置。以下为全量字段字典说明与生产级脱敏模板：

```toml
schema_version = 1

# ==============================================================================
# 1. 本端站点配置 (Local Site)
# ==============================================================================
[site]
id = "site-a"                         # 本端唯一站点标识符 (必填，全局不可重复)
networks = ["198.51.100.0/24"]        # 本端局域网网段 (CIDR 列表，用于向对端通告内网路由)
tunnel_address = "198.51.100.1/32"    # 本端 WireGuard IPv4 点对点虚拟隧道地址 (必填)
role = "proxy"                        # 角色: "proxy" (被动监听端/服务端) 或 "visitor" (主动发起端)
allow_relay = true                    # 是否允许在 P2P 打洞失败时使用 STCP 中继兜底 (默认 true)
mtu = 1420                            # WireGuard 接口 MTU (默认 1420，范围 576~9000，两端必须严格一致)

# 可选：IPv6 隧道与动态前缀同步配置 (Dual-Stack & Dynamic IPv6 Prefix)
[site.ipv6]
enabled = false
tunnel_address = "2001:db8:1::1/128"
lan_interfaces = ["br-lan"]
dynamic_source = false
max_active_prefixes = 16
max_history_prefixes = 256

# ==============================================================================
# 2. 对端站点配置 (Peer Site)
# ==============================================================================
# 单对端使用 [peer]；三台及以上 Hub-Spoke 网状组网改用 [[peers]] 数组依次声明多个对端。
[peer]
id = "site-b"                         # 对端站点标识符 (必填)
networks = ["198.51.200.0/24"]        # 对端局域网网段 (本端将自动安装指向这些网段的路由)
tunnel_address = "198.51.100.2/32"    # 对端 WireGuard IPv4 虚拟隧道地址 (必填)
role = "visitor"                      # 对端角色: 必须与本端互补 (本端 proxy 则对端 visitor，反之亦然)
allow_relay = true                    # 对端是否允许中继兜底 (默认 true)
mtu = 1420                            # 对端 MTU (范围 576~9000，两端必须严格一致)
public_key = "REPLACE_WITH_PEER_PUBLIC_KEY" # 对端 WireGuard 静态公钥 (Base64)
# listen_port = 51822                 # 可选：对端公网 UDP 监听端口 (省略时自动 = p2p.listen_port + 对端序号，多对端不得重复)
# shared_secret = "REPLACE_WITH_PEER_SHARED_SECRET" # 可选：本对端独立 P2P 预共享密钥 (省略时回退 frp.shared_secret)
# shared_secret_file = "/etc/frp-tun/peer-site-b-key" # 与 shared_secret 互斥

# 可选：私有中继出站候选 (本端作为 Spoke 经 Hub 站点中继，语义见"架构 - 私有中继")
# [peer.private_relay]
# hubs = ["site-hub"]                 # 最多 8 个候选 Hub 站点 ID (须为已配置的其他对端且非本对端自身)
# key = "REPLACE_WITH_RELAY_KEY"      # A-B 专用 32 字节 Base64 认证密钥 (不得复用 frp.shared_secret)
# key_file = "/etc/frp-tun/relay-site-b-key" # 与 key 互斥

# 可选：对端 IPv6 隧道与动态前缀接收配置
[peer.ipv6]
enabled = false
tunnel_address = "2001:db8:1::2/128"
accept_dynamic = false
sync_port = 51830

# ==============================================================================
# 3. FRP 信令与服务端配置 (FRP Signaling & Underlay)
# ==============================================================================
[frp]
server_addr = "relay.example.com"     # 核心必配：frps 服务端公网 IP 或域名 (必填)
server_port = 7000                    # 核心必配：frps 服务端口，通常为 7000 (必填)
tls = true                            # 是否与 frps 启用 TLS 加密通道 (默认 true)

# 凭据管理：内联字符串与独立文件路径互斥，生产部署强烈建议使用文件路径
token = "REPLACE_WITH_FRP_TOKEN"      # frps 身份验证 Token
# token_file = "/etc/frp-tun/frps-token"

shared_secret = "REPLACE_WITH_SHARED_SECRET" # 对等端 XTCP/STCP 预共享密钥 (全局默认值，可按对端覆盖；两端必须完全一致)
# shared_secret_file = "/etc/frp-tun/secret-key"

# 高级传输参数
client_id = ""                        # 实例标识 (留空则根据 MAC 地址自动派生)
transport_protocol = "quic"           # 与 frps 控制信令协议: "quic" (默认), "tcp", "kcp", "websocket", "wss"
# trusted_ca_file = "/etc/frp-tun/frps-ca" # 可选：自签名/私有 CA 场景下校验 frps TLS 证书的 CA 文件 (未配置时跳过证书有效期校验，仅输出告警)
p2p_use_compression = false           # XTCP P2P 直连数据压缩 (默认 false)
relay_use_compression = true          # STCP 中继兜底数据压缩 (默认 true)

# ==============================================================================
# 4. WireGuard 引擎配置 (WireGuard Underlay & Keys)
# ==============================================================================
[wireguard]
mode = "auto"                         # 驱动模式: "auto" (优先内核模块，失败平滑回退 userspace), "kernel", "userspace"
private_key = "REPLACE_WITH_PRIVATE_KEY" # 本机 WireGuard 私钥
# private_key_file = "/etc/frp-tun/identity-key" # 生产推荐：独立私钥文件路径 (权限须为 0600)
# interface_name = "ft0"              # 显式虚拟接口名 (留空则自动按 ft0, ft1... 分配)

# ==============================================================================
# 5. 内核转发与防火墙配置 (Forwarding & Firewall)
# ==============================================================================
[forwarding]
authorize_enable = true               # 授权守护进程自动开启系统内核转发
mode = "nftables"                     # 防火墙管理模式: "nftables" (默认，自动维护出入站规则) 或 "manual"

# ==============================================================================
# 6. P2P 打洞、双栈选路与 QoS (P2P Path & Multi-Path QoS)
# ==============================================================================
[p2p]
prefer_family = "ipv4"                # 直连优先级: "ipv4" (默认) 或 "ipv6"
listen_port = 51821                   # 固定公网 UDP 监听端口 (默认 51821)
manage_firewall = false               # 是否授权自动管理公网入站规则
data_transport = "auto"               # 直连数据面: "auto" (默认，QUIC Datagram，自动与旧版 Stream 对端协商互通) 或 "stream" (强制 Stream，兼容 v0.4.2)
nathole_coordinator = "frp"           # NAT 打洞首选协调机制: "frp" (默认，经 frps 信令协调) 或 "native" (本地 QUIC 直连引擎)；另一机制自动作为兜底阶段
nathole_frp_timeout = "10s"           # frp 协调阶段打洞预算 (默认 10s，范围 (0s,60s])
nathole_native_timeout = "10s"        # native 阶段打洞预算 (默认 10s，范围 (0s,60s])

# 可选：自定义 STUN 发现目标 (两种写法互斥；未配置时使用内置公共 STUN 列表，完整清单见下方"外连披露")
# stun_server = "stun.example.com:3478" # 单个自定义 STUN 目标 (置于内置列表之前优先探测)
# stun_servers = ["stun.example.com:3478", "stun-alt.example.com:3478"] # 目标列表 (最多 8 项；v0.5.2 仅校验未接入探测引擎)

port_mapping = true                   # 网关端口映射 (UPnP/PCP) 自动发现 (默认 true)
quic_keepalive_period = "10s"         # P2P QUIC 心跳保活周期 (默认 10s，范围 [0s,10m]；显式 "0s" 停用；旧键 keepalive_period 仍被接受)
accept_grace_after_failed_punch = "6s" # 打洞完全未命中后仍接收对端迟到拨入的宽限期 (默认 6s，范围 (0s,60s]；旧键 accept_grace 仍被接受)
family_standby = false                # 是否开启双栈温热备用

# 可选：多路径 QoS 测速与动态选路
[p2p.qos]
enabled = false                       # 是否开启动态 QoS 选路测速
probe_interval = "1s"                 # QoS 探测间隔 (范围 50ms~60s)
probe_timeout = "2s"                  # 探测超时阈值 (范围 50ms~60s)
rtt_ratio_threshold = 1.3             # RTT 倒换比例门槛 (范围 1.05~10.0)
rtt_min_diff = "15ms"                 # RTT 绝对差倒换门槛 (默认 15ms，范围 1ms~10s)
loss_diff_threshold = 0.05            # 丢包率差倒换门槛 (默认 0.05，设置大于 0 时启用，范围 0.01~1.0)
switchback_holddown = "30s"           # 优选路径回切观察期 (范围 1s~3600s)
# experimental_duplication = false    # 实验性：主备双路径同时复制发送流量，以带宽冗余对抗瞬时丢包

# ==============================================================================
# 7. Hub 私有中继服务端 (Private Relay Hub)
# ==============================================================================
# 本站点作为中继 Hub 时配置；Spoke 侧出站候选在 [peer.private_relay] 中声明。
[private_relay]
enabled = false                       # 是否允许为其他站点对承载端到端加密的 WireGuard 密文中继
pairs = []                            # 授权中继的站点对，如 [["site-a", "site-b"]] (要求 enabled = true，且两端均为已配置对端)
max_sessions = 8                      # 最大并发中继会话数 (范围 1~64)
queue_bytes = 65536                   # 每会话中继队列字节数 (范围 4096~4194304)
bytes_per_second = 8388608            # 每会话限速，字节/秒 (范围 16384~1073741824)

# ==============================================================================
# 8. 日志配置 (Logging)
# ==============================================================================
[log]
to = "console"                        # 日志输出去向: "console" 或指定文件路径
level = "info"                        # 日志级别: "trace", "debug", "info", "warn", "error"
max_days = 3                          # 日志保留天数
max_size = "10MB"                     # 单文件轮转大小
max_backups = 3                       # 保留历史切片数量
```

## 字段补充说明

- **STUN 发现目标**：未配置 `p2p.stun_server` 时，frp-tun 使用内置公共 STUN 目标列表完成 NAT 类型识别与公网端口映射探测（完整清单见下方外连披露），对称 NAT 端口预测同样依赖这些样本；配置后，自定义目标置于内置列表之前优先探测，内置目标继续作为兜底样本。`p2p.stun_servers` 列表键在 v0.5.2 中仅完成配置校验，尚未接入探测引擎（为向前兼容保留），实际探测目标由 `p2p.stun_server` 与内置列表共同决定；两个键不可同时配置。
- **打洞协调两阶段**：`p2p.nathole_coordinator` 只决定首选机制，另一机制会自动作为兜底阶段依次尝试；两阶段的预算分别由 `nathole_frp_timeout` 与 `nathole_native_timeout` 控制，单阶段超时后自动进入下一阶段。
- **多对端组网**：单对端使用 `[peer]`；三台及以上 Hub-Spoke 网状组网改用 `[[peers]]` 数组依次声明多个对端。公网监听端口默认按 `p2p.listen_port + 序号` 递增分配，也可用 `peer.listen_port` 显式指定（多对端之间不得重复）。
- **P2P 预共享密钥回退**：P2P 握手密钥按 `peer.shared_secret` → `peer.shared_secret_file` → `frp.shared_secret` → `frp.shared_secret_file` 的顺序解析，多对端部署可为每个对端配置独立密钥。
- **旧键名兼容**：`p2p.keepalive_period` 与 `p2p.accept_grace` 分别是 `p2p.quic_keepalive_period` 与 `p2p.accept_grace_after_failed_punch` 的旧别名，仅在新键未设置时生效。

## 内置 STUN 目标与外连披露

为便于安全审计与合规评估，此处公开 frp-tun 会主动连接的全部内置第三方地址。native 打洞阶段会向以下公共 STUN 服务器发送标准 STUN Binding 请求（UDP 3478 端口，域名经 DNS 解析为 IPv4 地址）：

| 内置目标 | 运营方 |
| --- | --- |
| `stun.miwifi.com:3478` | 小米 |
| `stun.chat.bilibili.com:3478` | 哔哩哔哩 |
| `stun.cloudflare.com:3478` | Cloudflare |
| `stun.easyvoip.com:3478` | 公共 STUN 服务 |

外连边界：

- 除上表外，frp-tun 不包含任何其他内置第三方外连：frps 信令与中继地址来自 `[frp].server_addr`，打洞直连与私有中继目标均为用户配置的对端站点；
- UPnP/PCP 端口映射发现仅与本局域网网关设备通信，不涉及公网第三方；
- 程序不包含遥测、版本检查或统计上报；
- 内置目标无法通过配置移除，严格外连管控的环境请在防火墙层面按上表放行或拦截相应域名。
